cmake_minimum_required(VERSION 3.20)
project(styio VERSION 0.0.1 LANGUAGES CXX)

# Configure-only contract fixture, not an LLVM or production-build substitute.
# It evaluates the real src target modules with fixed external inputs, allowing
# target properties to be compared on hosts with or without LLVM installed.
set(CMAKE_CXX_COMPILER_ID GNU)
set(MSVC FALSE)
if(STYIO_CONTRACT_MSVC)
  set(CMAKE_CXX_COMPILER_ID MSVC)
  set(MSVC TRUE)
endif()
set(CMAKE_DL_LIBS contract_dl)
set(CMAKE_INSTALL_BINDIR bin)
set(LLVM_LIBS contract_llvm)
set(STYIO_NANO_COMPILE_DEFINITIONS STYIO_CONTRACT_PROFILE=1)
foreach(name IN ITEMS LLVM_DIR SOURCE_DIR CMAKE_C_COMPILER CMAKE_CXX_COMPILER CMAKE_MAKE_PROGRAM)
  set(STYIO_DEFINE_${name} "contract_${name}")
endforeach()
function(styio_apply_llvm_compile_settings target_name)
  target_compile_definitions(${target_name} PRIVATE STYIO_CONTRACT_LLVM_SETTINGS)
endfunction()
if(STYIO_ENABLE_TREE_SITTER)
  set(tree_sitter_runtime_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/tree-sitter")
  add_library(styio_tree_sitter_runtime INTERFACE)
  add_library(styio_tree_sitter_styio INTERFACE)
endif()
add_subdirectory("${STYIO_CONTRACT_SOURCE_ROOT}/src" styio-src)
get_property(targets DIRECTORY "${STYIO_CONTRACT_SOURCE_ROOT}/src" PROPERTY BUILDSYSTEM_TARGETS)
list(SORT targets)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/target-properties.txt" "")
foreach(target IN LISTS targets)
  foreach(property IN ITEMS TYPE SOURCES LINK_LIBRARIES INTERFACE_LINK_LIBRARIES
      INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES COMPILE_DEFINITIONS
      COMPILE_OPTIONS OUTPUT_NAME MANUALLY_ADDED_DEPENDENCIES)
    get_target_property(value ${target} ${property})
    if(value STREQUAL "value-NOTFOUND")
      set(value "")
    endif()
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/target-properties.txt" "${target}|${property}|${value}\n")
  endforeach()
endforeach()
