@styio_str = private unnamed_addr constant [27 x i8] c"/tmp/styio_pipeline_in.txt\00", align 1

define i32 @main() {
main_entry:
  %0 = call i64 @styio_file_open(ptr @styio_str)
  %f = alloca i64, align 8
  store i64 %0, ptr %f, align 8
  %1 = call i32 @styio_runtime_has_error()
  %2 = icmp ne i32 %1, 0
  br i1 %2, label %runtime_fail, label %runtime_ok

runtime_fail:                                     ; preds = %main_entry
  %3 = load i64, ptr %f, align 8
  call void @styio_file_close(i64 %3)
  store i64 0, ptr %f, align 8
  ret i32 0

runtime_ok:                                       ; preds = %main_entry
  %4 = load i64, ptr %f, align 8
  call void @styio_file_rewind(i64 %4)
  br label %fline_hdr

fline_hdr:                                        ; preds = %fline_body, %runtime_ok
  %5 = load i64, ptr %f, align 8
  %6 = call ptr @styio_file_read_line(i64 %5)
  %7 = icmp eq ptr %6, null
  br i1 %7, label %fline_exit, label %fline_body

fline_body:                                       ; preds = %fline_hdr
  %line = alloca ptr, align 8
  store ptr %6, ptr %line, align 8
  %8 = load ptr, ptr %line, align 8
  call void @styio_stdout_write_cstr(ptr %8)
  br label %fline_hdr

fline_exit:                                       ; preds = %fline_hdr
  %9 = load i64, ptr %f, align 8
  call void @styio_file_close(i64 %9)
  store i64 0, ptr %f, align 8
  ret i32 0
}
