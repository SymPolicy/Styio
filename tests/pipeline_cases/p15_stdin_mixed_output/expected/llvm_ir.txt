; ModuleID = 'styio'
source_filename = "styio"
@styio_str = private unnamed_addr constant [6 x i8] c"got: \00", align 1
@styio_str.1 = private unnamed_addr constant [13 x i8] c"processing: \00", align 1
define i32 @main() {
main_entry:
  %line = alloca ptr, align 8
  br label %stdin_hdr
stdin_hdr:                                        ; preds = %stdin_body, %main_entry
  %tmp = call ptr @styio_stdin_read_line()
  %tmp = icmp eq ptr %tmp, null
  br i1 %tmp, label %stdin_exit, label %stdin_body
stdin_body:                                       ; preds = %stdin_hdr
  store ptr %tmp, ptr %line, align 8
  %tmp = load ptr, ptr %line, align 8
  %tmp = call ptr @styio_strcat_ab(ptr @styio_str, ptr %tmp)
  ; STYIO_STDOUT_CSTR %tmp
  call void @styio_free_cstr(ptr %tmp)
  %tmp = load ptr, ptr %line, align 8
  %tmp = call ptr @styio_strcat_ab(ptr @styio_str.1, ptr %tmp)
  call void @styio_stderr_write_cstr(ptr %tmp)
  call void @styio_free_cstr(ptr %tmp)
  br label %stdin_hdr
stdin_exit:                                       ; preds = %stdin_hdr
  ret i32 0
}
