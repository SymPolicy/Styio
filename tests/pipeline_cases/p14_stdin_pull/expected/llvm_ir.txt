; ModuleID = 'styio'
source_filename = "styio"
define i32 @main() {
main_entry:
  %x = alloca i64, align 8
  %tmp = call ptr @styio_stdin_read_line()
  %tmp = call i64 @styio_cstr_to_i64(ptr %tmp)
  %tmp = call i32 @styio_runtime_has_error()
  %tmp = icmp ne i32 %tmp, 0
  br i1 %tmp, label %runtime_fail, label %runtime_ok
runtime_fail:                                     ; preds = %main_entry
  ret i32 0
runtime_ok:                                       ; preds = %main_entry
  store i64 %tmp, ptr %x, align 8
  %tmp = load i64, ptr %x, align 8
  ; STYIO_STDOUT_I64 %tmp
  ret i32 0
}
