; ModuleID = 'styio'
source_filename = "styio"
define i32 @main() {
main_entry:
  %result = alloca i64, align 8
  %line = alloca ptr, align 8
  br label %stdin_hdr
stdin_hdr:                                        ; preds = %runtime_ok, %main_entry
  %tmp = call ptr @styio_stdin_read_line()
  %tmp = icmp eq ptr %tmp, null
  br i1 %tmp, label %stdin_exit, label %stdin_body
stdin_body:                                       ; preds = %stdin_hdr
  store ptr %tmp, ptr %line, align 8
  %tmp = load ptr, ptr %line, align 8
  %tmp = call i64 @styio_cstr_to_i64(ptr %tmp)
  %tmp = call i32 @styio_runtime_has_error()
  %tmp = icmp ne i32 %tmp, 0
  br i1 %tmp, label %runtime_fail, label %runtime_ok
stdin_exit:                                       ; preds = %stdin_hdr
  ret i32 0
runtime_fail:                                     ; preds = %stdin_body
  ret i32 0
runtime_ok:                                       ; preds = %stdin_body
  %tmp = mul i64 %tmp, 2
  store i64 %tmp, ptr %result, align 8
  %tmp = load i64, ptr %result, align 8
  ; STYIO_STDOUT_I64 %tmp
  br label %stdin_hdr
}
