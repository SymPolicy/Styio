; ModuleID = 'styio'
source_filename = "styio"
@styio_str = private unnamed_addr constant [51 x i8] c"tests/pipeline_cases/p05_snapshot_accum/factor.txt\00", align 1
@styio_fe_lit = private constant [3 x i64] [i64 1, i64 2, i64 3]
define i32 @main() {
main_entry:
  %prod = alloca i64, align 8
  %x = alloca i64, align 8
  %fe_idx = alloca i64, align 8
  %sum = alloca i64, align 8
  %factor = alloca i64, align 8
  %tmp = call i64 @styio_read_file_i64line(ptr @styio_str)
  store i64 %tmp, ptr %factor, align 8
  store i64 0, ptr %sum, align 8
  store i64 0, ptr %fe_idx, align 8
  br label %foreach_hdr
foreach_exit:                                     ; preds = %foreach_hdr
  %tmp = load i64, ptr %sum, align 8
  ; STYIO_STDOUT_I64 %tmp
  ret i32 0
foreach_hdr:                                      ; preds = %foreach_step, %main_entry
  %tmp = load i64, ptr %fe_idx, align 8
  %tmp = icmp slt i64 %tmp, 3
  br i1 %tmp, label %foreach_body, label %foreach_exit
foreach_body:                                     ; preds = %foreach_hdr
  %tmp = load i64, ptr %fe_idx, align 8
  %tmp = getelementptr inbounds [3 x i64], ptr @styio_fe_lit, i32 0, i64 %tmp
  %tmp = load i64, ptr %tmp, align 8
  store i64 %tmp, ptr %x, align 8
  %tmp = load i64, ptr %x, align 8
  %tmp = load i64, ptr %factor, align 8
  %tmp = mul i64 %tmp, %tmp
  store i64 %tmp, ptr %prod, align 8
  %tmp = load i64, ptr %sum, align 8
  %tmp = load i64, ptr %prod, align 8
  %tmp = add i64 %tmp, %tmp
  store i64 %tmp, ptr %sum, align 8
  br label %foreach_step
foreach_step:                                     ; preds = %foreach_body
  %tmp = load i64, ptr %fe_idx, align 8
  %tmp = add i64 %tmp, 1
  store i64 %tmp, ptr %fe_idx, align 8
  br label %foreach_hdr
}
